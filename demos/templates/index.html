KQL Query Showcase - Real-Time Analytics

Sample KQL Queries

// Top repositories by activity
github_events
| summarize count() by repo_name
| top 10 by count_
| order by count_ desc
// Collaboration patterns over time
github_events
| where event_type in ('PushEvent', 'PullRequestEvent')
| summarize events=count() by bin(created_at, 1h), event_type
| render timechart
// Developer activity heatmap
github_events
| summarize count() by actor_login, dayofweek(created_at)
| evaluate pivot(actor_login, dayofweek_created_at, sum(count_))
| render heatmap
